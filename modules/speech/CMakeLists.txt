# 本模块可独立配置，不依赖主程序的 Windows 构建门控。
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.21)
    project(coral_speech_standalone LANGUAGES CXX)
    include(CTest)
endif()
get_filename_component(CORAL_SPEECH_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    option(CORAL_SPEECH_BUNDLED_CURL "Build the vendored curl instead of finding a system package" OFF)
    if(CORAL_SPEECH_BUNDLED_CURL)
        set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
        set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
        set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
        set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
        set(CURL_USE_LIBSSH2 OFF CACHE BOOL "" FORCE)
        set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
        set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
        set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
        set(CURL_USE_NGHTTP2 OFF CACHE BOOL "" FORCE)
        set(CURL_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
        if(WIN32)
            set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
            set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
        endif()
        function(coral_speech_add_curl)
            # 仓库只保留 curl 源码子集，关闭其测试且不影响本模块测试。
            set(BUILD_TESTING OFF)
            add_subdirectory("${CORAL_SPEECH_ROOT}/src/lib/curl" curl EXCLUDE_FROM_ALL)
        endfunction()
        coral_speech_add_curl()
    endif()
endif()
add_library(coral_speech STATIC audio_clip.cpp tencent_protocol.cpp tencent_recognizer.cpp curl_transport.cpp)
add_library(coral::speech ALIAS coral_speech)
target_compile_features(coral_speech PUBLIC cxx_std_17)
find_package(Threads REQUIRED)
target_link_libraries(coral_speech PRIVATE Threads::Threads)
target_include_directories(coral_speech PUBLIC "${CORAL_SPEECH_ROOT}/include" PRIVATE
    "${CORAL_SPEECH_ROOT}/third_party/picosha2"
    "${CORAL_SPEECH_ROOT}/src/lib/opencc/deps/rapidjson-1.1.0")
if(TARGET libcurl_static)
    target_link_libraries(coral_speech PRIVATE libcurl_static)
else()
    find_package(CURL 7.85 REQUIRED)
    target_link_libraries(coral_speech PRIVATE CURL::libcurl)
endif()
if(MSVC)
    target_compile_options(coral_speech PRIVATE /utf-8 /Zc:__cplusplus)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_subdirectory("${CORAL_SPEECH_ROOT}/tools/native/speech_probe" speech_probe)
    if(BUILD_TESTING)
        add_subdirectory("${CORAL_SPEECH_ROOT}/tests/speech" speech_tests)
    endif()
endif()
