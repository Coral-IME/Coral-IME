cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(fuzzy_dialog_tests LANGUAGES CXX)
    enable_testing()
endif()
add_executable(fuzzy_dialog_smoke fuzzy_dialog_smoke.cpp)
target_compile_features(fuzzy_dialog_smoke PRIVATE cxx_std_17)
target_include_directories(fuzzy_dialog_smoke PRIVATE support
    ../../src/platform/windows/renderer)
target_compile_definitions(fuzzy_dialog_smoke PRIVATE UNICODE _UNICODE NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(fuzzy_dialog_smoke PRIVATE /utf-8)
endif()
target_link_libraries(fuzzy_dialog_smoke PRIVATE user32 gdi32 advapi32)
add_test(NAME fuzzy_dialog_smoke COMMAND fuzzy_dialog_smoke "${CMAKE_CURRENT_BINARY_DIR}/renders")
