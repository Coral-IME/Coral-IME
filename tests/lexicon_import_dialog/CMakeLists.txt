cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(lexicon_import_dialog_tests LANGUAGES CXX)
    enable_testing()
endif()
add_executable(lexicon_import_dialog_smoke lexicon_import_dialog_smoke.cpp)
if(TARGET coral::lexicon_io)
    target_link_libraries(lexicon_import_dialog_smoke PRIVATE coral::lexicon_io)
else()
    target_sources(lexicon_import_dialog_smoke PRIVATE ../../platform/windows/lexicon_io/text_import.cpp)
endif()
target_compile_features(lexicon_import_dialog_smoke PRIVATE cxx_std_17)
target_include_directories(lexicon_import_dialog_smoke PRIVATE support
    ../../include
    ../../src/platform/windows/renderer)
target_compile_definitions(lexicon_import_dialog_smoke PRIVATE UNICODE _UNICODE NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(lexicon_import_dialog_smoke PRIVATE /utf-8)
endif()
target_link_libraries(lexicon_import_dialog_smoke PRIVATE user32 gdi32 advapi32 comctl32 comdlg32 shell32)
add_test(NAME lexicon_import_dialog_smoke COMMAND lexicon_import_dialog_smoke "${CMAKE_CURRENT_BINARY_DIR}/renders")
